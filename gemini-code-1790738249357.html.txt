    <script>

        function joinGroupRoom() {

            if (roomId.length !== 4 || isNaN(roomId)) {

        // ルーム初期化（中継ハブ方式対応）
        function initRoom(modeName, roomId) {
            if (peer) {
                peer.destroy();
            }
            connections = {};
            incomingBuffers = {};
            currentRoomId = roomId;
            isHost = false;

            document.getElementById('chat-room-title').innerText = `${modeName} (コード: ${currentRoomId})`;
            document.getElementById('chat-room-subtitle').innerText = '接続確立中...';
            document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-blue-500 animate-pulse"></span><span>接続中</span>';

            clearMessages();
            appendSystemMessage(`ルーム [コード: ${currentRoomId}] へ接続しています...`);

            const hostPeerId = `nexus_room_${roomId}_host`;
            const myGuestId = `nexus_room_${roomId}_guest_${Math.random().toString(36).substring(2, 6)}`;

            // ホストになれるか試す
            peer = new Peer(hostPeerId);

            peer.on('open', (id) => {
                isHost = true;
                document.getElementById('chat-room-subtitle').innerText = 'ホストとして運営中（参加者待受）';
                appendSystemMessage('あなたがこのルームの「ホスト(中継ハブ)」になりました。');
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-emerald-500"></span><span>ホスト稼働中</span>';
                
                peer.on('connection', (conn) => {
                    setupHostConnection(conn);
                });
            });

            peer.on('error', (err) => {
                if (err.type === 'unavailable-id') {
                    // すでにホストが居るので、ゲストとして参加
                    peer.destroy();
                    isHost = false;

                    peer = new Peer(myGuestId);
                    peer.on('open', () => {
                        document.getElementById('chat-room-subtitle').innerText = 'ゲストとして接続完了';
                        appendSystemMessage('ホストに接続しました。');
                        const conn = peer.connect(hostPeerId);
                        setupGuestConnection(conn);
                    });

                    peer.on('error', (subErr) => {
                        console.error(subErr);
                        document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-red-500"></span><span>エラー</span>';
                        appendSystemMessage(`接続エラー: ${subErr.type}`);
                    });
                } else {
                    console.error(err);
                    document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-red-500"></span><span>エラー</span>';
                    appendSystemMessage(`エラー: ${err.type}`);
                }
            });
        }

        // ホスト側：ゲストからの接続を受け付け、他のゲスト全員へ転送する（中継）
        function setupHostConnection(conn) {
            connections[conn.peer] = conn;

            conn.on('open', () => {
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-emerald-500"></span><span>接続済み (参加者あり)</span>';
                appendSystemMessage('新しいメンバーが参加しました！');
                updatePeerCount();
            });

            conn.on('close', () => {
                delete connections[conn.peer];
                appendSystemMessage('メンバーが退出しました');
                updatePeerCount();
            });

            conn.on('data', (packet) => {
                // ホストが受け取ったメッセージを、自分以外の全接続ピア（他のゲスト）に転送する
                broadcastToOthers(packet, conn.peer);
                handleDataPacket(packet);
            });
        }

        // ゲスト側：ホストと接続し、メッセージを送受信する
        function setupGuestConnection(conn) {
            connections[conn.peer] = conn;

            conn.on('open', () => {
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-emerald-500"></span><span>接続済み</span>';
                appendSystemMessage('ルームへの接続に成功しました！');
                updatePeerCount();
            });

            conn.on('close', () => {
                delete connections[conn.peer];
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-amber-500"></span><span>切断</span>';
                appendSystemMessage('ホストとの接続が切れました');
                updatePeerCount();
            });

            conn.on('data', (packet) => {
                handleDataPacket(packet);
            });
        }

        // ホストが自分以外のメンバーへパケットを転送する関数
        function broadcastToOthers(packet, senderPeerId) {
            Object.keys(connections).forEach(peerId => {
                if (peerId !== senderPeerId) {
                    const conn = connections[peerId];
                    if (conn && conn.open) {
                        conn.send(packet);
                    }
                }
            });
        }

        // 全ピアへメッセージを送信する共通関数
        function sendPayloadToAll(payloadObj) {
            Object.keys(connections).forEach(peerId => {
                const conn = connections[peerId];
                if (conn && conn.open) {
                    conn.send(payloadObj);
                }
            });
        }

        function handleDataPacket(packet) {
            if (packet.type === 'chat') {
                appendMessage(packet.senderName, packet.senderAvatar, packet.text, false, packet.timestamp, null, packet.isSticker, true);
            } else if (packet.type === 'image_full') {
                appendMessage(packet.senderName, packet.senderAvatar, '', false, packet.timestamp, packet.image, false, true);
            }
        }

        function updatePeerCount() {
            const count = Object.keys(connections).filter(id => connections[id].open).length;
            document.getElementById('peer-count').innerText = `接続ピア: ${count}`;
            if (count > 0) {
                document.getElementById('chat-room-subtitle',).innerText = isHost ? `ホスト運営中 (参加者 ${count}名)` : `接続中 (同期完了)`;
            }
        }

        function sendMessage() {
            const input = document.getElementById('message-input');
            const text = input.value.trim();
            if (!text) return;

            const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            appendMessage(myName, myAvatar, text, true, timestamp, null, false, true);
            
            const payload = {
                type: 'chat',
                senderName: myName,
                senderAvatar: myAvatar,
                text: text,
                timestamp: timestamp,
                isSticker: false
            };

            sendPayloadToAll(payload);
            input.value = '';
        }

        function sendSticker(stickerText) {
            toggleStickerPanel();
            const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            appendMessage(myName, myAvatar, stickerText, true, timestamp, null, true, true);

            const payload = {
                type: 'chat',
                senderName: myName,
                senderAvatar: myAvatar,
                text: stickerText,
                timestamp: timestamp,
                isSticker: true
            };

            sendPayloadToAll(payload);
        }

        function sendImage(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const base64Image = e.target.result;
                const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

                appendMessage(myName, myAvatar, '', true, timestamp, base64Image, false, true);

                const payload = {
                    type: 'image_full',
                    senderName: myName,
                    senderAvatar: myAvatar,
                    timestamp: timestamp,
                    image: base64Image
                };

                sendPayloadToAll(payload);
            };
            reader.readAsDataURL(file);
        }

        function handleKeyPress(e) {
            if (e.key === 'Enter') {
                sendMessage();
            }
        }

        // メッセージ表示 ＆ 履歴への自動保存
        function appendMessage(sender, senderAvatar, text, isMe, timestamp, image = null, isSticker = false, saveToLog = true) {
            const container = document.getElementById('messages-container');
            const messageRow = document.createElement('div');
            messageRow.className = `flex items-end space-x-2 mb-3 ${isMe ? 'justify-end' : 'justify-start'}`;

            let avatarHtml = '';
            if (senderAvatar) {
                avatarHtml = `<div class="w-8 h-8 rounded-full overflow-hidden flex-shrink-0 shadow border border-indigo-500"><img src="${senderAvatar}" class="w-full h-full object-cover"></div>`;
            } else {
                const initial = sender ? sender.charAt(0) : 'U';
                avatarHtml = `<div class="w-8 h-8 bg-indigo-600 text-white rounded-full flex items-center justify-center font-bold text-xs flex-shrink-0 shadow">${initial}</div>`;
            }

            let contentHtml = '';
            if (image) {
                contentHtml += `<img src="${image}" class="max-w-xs rounded-xl mb-1 shadow border border-slate-700">`;
            }
            if (text) {
                if (isSticker) {
                    contentHtml += `<div class="text-sm font-bold py-1.5 px-3 bg-indigo-950 text-indigo-200 border border-indigo-700/50 rounded-xl shadow-inner">${escapeHtml(text)}</div>`;
                } else {
                    contentHtml += `<p class="text-sm text-slate-100">${escapeHtml(text)}</p>`;
                }
            }

            if (isMe) {
                messageRow.innerHTML = `
                    <div class="flex items-end space-x-1 max-w-xs md:max-w-md">
                        <span class="text-[10px] text-slate-400 self-end mb-1">${timestamp}</span>
                        <div class="bg-indigo-600 text-white rounded-2xl px-4 py-2.5 shadow-md rounded-br-none">
                            ${contentHtml}
                        </div>
                    </div>
                    ${avatarHtml}
                `;
            } else {
                messageRow.innerHTML = `
                    ${avatarHtml}
                    <div class="flex flex-col">
                        <span class="text-xs text-slate-400 ml-1 mb-1">${escapeHtml(sender)}</span>
                        <div class="flex items-end space-x-1 max-w-xs md:max-w-md">
                            <div class="bg-slate-800 border border-slate-700 text-slate-100 rounded-2xl px-4 py-2.5 shadow-md rounded-bl-none">
                                ${contentHtml}
                            </div>
                            <span class="text-[10px] text-slate-400 self-end mb-1">${timestamp}</span>
                        </div>
                    </div>
                `;
            }

            container.appendChild(messageRow);
            container.scrollTop = container.scrollHeight;

            // 会話履歴ログへの保存
            if (saveToLog) {
                globalChatHistory.push({
                    sender: sender,
                    senderAvatar: senderAvatar,
                    text: text || (image ? '[画像]' : ''),
                    timestamp: timestamp,
                    room: currentRoomId || '未所属',
                    date: new Date().toLocaleDateString()
                });
                // 最大200件まで保持
                if (globalChatHistory.length > 200) {
                    globalChatHistory.shift();
                }
                localStorage.setItem('nexus_chat_history', JSON.stringify(globalChatHistory));
            }
        }

        function appendSystemMessage(text) {
            const container = document.getElementById('messages-container');
            const div = document.createElement('div');
            div.className = 'text-center my-2';
            div.innerHTML = `<span class="bg-slate-800 border border-slate-700 text-slate-400 text-xs px-3 py-1 rounded-full">${escapeHtml(text)}</span>`;
            container.appendChild(div);
            container.scrollTop = container.scrollHeight;
        }

        function clearMessages() {
            document.getElementById('messages-container').innerHTML = '';
        }

        function renderHistoryList() {
            const listContainer = document.getElementById('history-content-list');
            if (globalChatHistory.length === 0) {
                listContainer.innerHTML = `<p class="text-xs text-slate-500 text-center py-6">まだ記録された会話はありません。</p>`;
                return;
            }

            listContainer.innerHTML = '';
            // 新しい順に並び替え
            const reversed = [...globalChatHistory].reverse();
            reversed.forEach(item => {
                const div = document.createElement('div');
                div.className = 'bg-slate-800 border border-slate-700 rounded-xl p-3 space-y-1 text-xs';
                div.innerHTML = `
                    <div class="flex justify-between text-slate-400 text-[10px]">
                        <span class="font-bold text-indigo-400">${escapeHtml(item.sender)}</span>
                        <span>ルーム: ${escapeHtml(item.room)} | ${escapeHtml(item.date)} ${escapeHtml(item.timestamp)}</span>
                    </div>
                    <p class="text-slate-200 font-medium">${escapeHtml(item.text)}</p>
                `;
                listContainer.appendChild(div);
            });
        }

        function clearAllHistory() {
            if (confirm('すべての会話履歴を消去してもよろしいですか？')) {
                globalChatHistory = [];
                localStorage.removeItem('nexus_chat_history');
                renderHistoryList();
            }
        }

        function showRoomInfo() {
            if (currentRoomId) {
                alert(`Nexus Chat セッション情報\nコード: ${currentRoomId}\n役割: ${isHost ? 'ホスト(中継サーバー)' : 'ゲスト'}\n記録された総ログ数: ${globalChatHistory.length}件`);
            } else {
                alert('アクティブなセッションはありません。');
            }
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
        }
